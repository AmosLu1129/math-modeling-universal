\begin{algorithm}[H]
\caption{Template Iterative Solver Scheme}
\label{alg:template_solver}
\begin{algorithmic}[1]
\Require Parameter vector $\theta_0$, tolerance $\varepsilon > 0$, maximum iterations $K_{\max}$
\Ensure Converged state trajectory $x^*$ and objective value $J^*$
\For{$k = 1, \dots, K_{\max}$}
  \State Evaluate governing residual $r_k \gets \mathcal{F}(x_k; \theta_0)$
  \State Update state $x_{k+1} \gets \Phi(x_k, r_k)$
  \If{$\|r_k\|_2 < \varepsilon$}
    \State \textbf{break}
  \EndIf
\EndFor
\State \Return $x_k, J(x_k)$
\end{algorithmic}
\end{algorithm}
